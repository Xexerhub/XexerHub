-- Shared Utility Functions
local function formatTree(instance, depth)
    local indent = string.rep("    ", depth)
    local branch = depth > 0 and "|-- " or ""
    return string.format("%s%s[%s] %s", indent, branch, instance.ClassName, instance.Name)
end

local function getCleanPath(inst)
    local parts = {}
    local curr = inst
    while curr and curr ~= game do
        local name = curr.Name
        if name:match("^[%a_][%w_]*$") then
            table.insert(parts, 1, "." .. name)
        else
            table.insert(parts, 1, '["' .. name:gsub('"', '\\"') .. '"]')
        end
        curr = curr.Parent
    end
    if #parts > 0 and parts[1]:sub(1, 1) == "." then
        parts[1] = parts[1]:sub(2)
    end
    local serviceName = inst:GetFullName():split(".")[1]
    return "game:GetService(\"" .. serviceName .. "\")." .. table.concat(parts, ""):gsub("^[^%.]+%.?", "")
end

local function fireRemote(remote, ...)
    if remote:IsA("RemoteEvent") or remote:IsA("UnreliableRemoteEvent") then
        remote:FireServer(...)
    elseif remote:IsA("RemoteFunction") then
        return remote:InvokeServer(...)
    end
end
]])

            table.insert(fullOutput, "--------------------------------------------------")
            table.insert(fullOutput, "local remotes = {}")
            for i, rem in ipairs(remotesFound) do

                    table.insert(fullOutput, string.format("remotes[%d] = %s -- Class: %s", i, getCleanPath(rem), rem.ClassName))

            table.insert(fullOutput, "\n--------------------------------------------------")
            table.insert(fullOutput, "-- 3. CONFIG & DATA MODULES (" .. #modulesFound .. " Modules)")

local function dumpAllRemotes()
    if isDumpingRemotes then
        WindUI:Notify({ Title = "Warning", Content = "A remote scan is already in progress.", Icon = "alert-triangle", Duration = 3 })
        return
    end

    isDumpingRemotes = true
    setRemoteStatus("Scanning all remotes...")

    task.spawn(function()
        local startTime = os.clock()
        local remotesFound = {}
        local fullOutput = {}

        local servicesToScan = {
            game:GetService("ReplicatedStorage"),
            game:GetService("Workspace"),
            game:GetService("Lighting"),
            game:GetService("StarterGui"),
            game:GetService("SoundService"),
            game:GetService("MaterialService")
        }

        if LP and LP:FindFirstChild("PlayerScripts") then table.insert(servicesToScan, LP.PlayerScripts) end
        if LP and LP.Character then table.insert(servicesToScan, LP.Character) end

        for _, svc in ipairs(servicesToScan) do
            pcall(function()
                for _, obj in ipairs(svc:GetDescendants()) do
                    if obj:IsA("RemoteEvent") or obj:IsA("RemoteFunction") or obj:IsA("UnreliableRemoteEvent") then
                        table.insert(remotesFound, obj)
                    end
                end
            end)
        end

        table.insert(fullOutput, "==================================================")
        table.insert(fullOutput, "           XEXER ALL REMOTES DUMP REPORT          ")
        table.insert(fullOutput, "==================================================")
        table.insert(fullOutput, string.format("Place ID        : %s", tostring(game.PlaceId)))
        table.insert(fullOutput, string.format("Job ID          : %s", game.JobId))
        table.insert(fullOutput, string.format("Date & Time     : %s", os.date("%Y-%m-%d %H:%M:%S")))
        table.insert(fullOutput, string.format("Total Remotes   : %d items", #remotesFound))
        table.insert(fullOutput, "==================================================\n")

        table.insert(fullOutput, "-- [Discovered Remotes & Usage Syntax]")
        table.insert(fullOutput, "--------------------------------------------------")

        for i, rem in ipairs(remotesFound) do
            pcall(function()
                local path = getCleanPath(rem)
                local callSyntax = rem:IsA("RemoteFunction") and ":InvokeServer(...)" or ":FireServer(...)"
                table.insert(fullOutput, string.format("[%d] %s", i, rem.Name))
                table.insert(fullOutput, string.format("    Class : %s", rem.ClassName))
                table.insert(fullOutput, string.format("    Path  : %s", path))
                table.insert(fullOutput, string.format("    Usage : %s%s\n", path, callSyntax))
            end)
        end

        local elapsed = string.format("%.2f", os.clock() - startTime)
        table.insert(fullOutput, "==================================================")
        table.insert(fullOutput, string.format("Scan completed in : %s seconds", elapsed))
        table.insert(fullOutput, "==================================================")

        local fileName = string.format("Xexer_AllRemotes_%s.txt", tostring(game.PlaceId))

        setRemoteStatus("Writing .txt file to workspace...")
        if writefile then
            writefile(fileName, table.concat(fullOutput, "\n"))
            setRemoteStatus(string.format("Done! Saved %s (%d Remotes)", fileName, #remotesFound))
            WindUI:Notify({ Title = "Dump Completed!", Content = string.format("Saved %s (%d items)", fileName, #remotesFound), Icon = "check", Duration = 5 })
        else
            setRemoteStatus("Failed: writefile is unsupported")
            WindUI:Notify({ Title = "Error", Content = "Your executor does not support writefile.", Icon = "x", Duration = 4 })
        end

        isDumpingRemotes = false
    end)
end

            WindUI:Notify({ Title = "Warning", Content = "A dump process is already running.", Icon = "alert-triangle", Duration = 3 })
            return
        end

        SettingsExplorer.IsDumping = true
        _G.OnDumpStatusExplorer("Analyzing target hierarchy...")

        task.spawn(function()
            local targetRoots = {}
            local selected = SettingsExplorer.Target

            if selected == "[All Services - Full Dump]" then
                for _, s in ipairs(game:GetChildren()) do
                    pcall(function() table.insert(targetRoots, s) end)
                end
            elseif selected == "PlayerScripts" then
                if LP and LP:FindFirstChild("PlayerScripts") then table.insert(targetRoots, LP.PlayerScripts) end
            elseif selected == "LocalPlayer Character" then
                if LP and LP.Character then table.insert(targetRoots, LP.Character) end
            else
                local found = game:FindFirstChild(selected)
                if found then table.insert(targetRoots, found) end
            end

            if #targetRoots == 0 then
                _G.OnDumpStatusExplorer("Error: Target instance not found.")
                SettingsExplorer.IsDumping = false
                return
            end

            local startTime = os.clock()
            local output = {}

            table.insert(output, "==================================================")
            table.insert(output, "               XEXER DATA DUMP REPORT             ")
            table.insert(output, "==================================================")
            table.insert(output, string.format("Selected Target : %s", selected))
            table.insert(output, string.format("Place ID        : %s", tostring(game.PlaceId)))
            table.insert(output, string.format("Job ID          : %s", game.JobId))
            table.insert(output, string.format("Date & Time     : %s", os.date("%Y-%m-%d %H:%M:%S")))
            table.insert(output, string.format("Decompile Source: %s", SettingsExplorer.IncludeSource and "Enabled" or "Disabled"))
            table.insert(output, "==================================================\n")

            local totalObjects = 0
            local totalModules = 0

            local function scanHierarchy(parentInstance, depth)
                for _, child in ipairs(parentInstance:GetChildren()) do
                    totalObjects = totalObjects + 1
                    table.insert(output, formatTree(child, depth))

                    if SettingsExplorer.IncludeSource and child:IsA("ModuleScript") then
                        totalModules = totalModules + 1
                        local indent = string.rep("    ", depth + 1)
                        table.insert(output, indent .. "/* [SOURCE CODE START] */")
                        local success, code = pcall(function()
                            return decompile and decompile(child) or "-- Executor does not support decompile."
                        end)
                        if success and code and code ~= "" then
                            for line in string.gmatch(code, "[^\r\n]+") do
                                table.insert(output, indent .. line)
                            end
                        else
                            table.insert(output, indent .. "-- [Error: Unable to read script source]")
                        end
                        table.insert(output, indent .. "/* [SOURCE CODE END] */\n")
                    end
                    scanHierarchy(child, depth + 1)
                end
            end

            for _, root in ipairs(targetRoots) do
                pcall(function()
                    _G.OnDumpStatusExplorer("Scanning: " .. root.Name .. "...")
                    task.wait(0.01)
                    table.insert(output, string.format("\n[ ROOT SERVICE: %s ]", root.Name))
                    table.insert(output, string.rep("-", 45))
                    scanHierarchy(root, 0)
                end)
            end

            local elapsed = string.format("%.2f", os.clock() - startTime)
            table.insert(output, "\n==================================================")
            table.insert(output, string.format("Scan completed in : %s seconds", elapsed))
            table.insert(output, string.format("Total instances   : %d objects", totalObjects))
            table.insert(output, string.format("Modules decompiled: %d objects", totalModules))
            table.insert(output, "==================================================")

            local safeName = selected:gsub("%W", "")

                WindUI:Notify({ Title = "Error", Content = "Your executor does not support writefile.", Icon = "x", Duration = 4 })
